# Core dependencies. Python 3.11.
#
# CPU-safe by design: these install and run anywhere, including the author's laptop,
# which has no NVIDIA GPU. CUDA pins live in requirements-cuda.txt and are layered on
# top for whatever training box is used. See CLAUDE.md section 4.
#
# Versions are pinned exactly. A thesis number that cannot be reproduced because a
# transitive dependency moved is not a thesis number.
#
# This file grows per stage rather than front-loading every dependency the project will
# eventually need; each addition is tied to the stage that justified it.

# --- Stage 0: canonical format, config, seeding, run logging ---
numpy==1.26.4            # <2.0 deliberately: the mmcv/MMPose stack needed at Stage 3
                         # does not yet build against numpy 2.x

PyYAML==6.0.2            # classes.txt sidecar metadata, meta.yaml
omegaconf==2.3.0         # config composition and dotlist overrides

tensorboard==2.18.0      # Hard Rule 4, alternative backend

# --- Stage 3: feature extraction ---
# Pillow decodes the extracted JPEG frames. It is imported lazily by the frame source, so
# the causal machinery and its tests still run without it.
Pillow==11.0.0
# transformers pulls the VideoMAEv2 checkpoint. CPU-safe: it installs and imports fine
# without CUDA, it is simply slow. The GPU pins live in requirements-cuda.txt.
transformers==4.46.3
safetensors==0.4.5
# The checkpoint's remote modeling code (trust_remote_code) imports these two; transformers
# refuses to load it without them. Found on the first real load, 2026-10-04 -- the test
# suite never executes the checkpoint, so it could not catch the gap. timm supplies only
# drop_path / to_2tuple / trunc_normal_, all inert at eval with loaded weights, so its
# version cannot change the features.
timm==1.0.30
easydict==1.13
# timm requires torchvision. Pinned to the release paired with the CPU torch above;
# requirements-cuda.txt replaces both with the CUDA pair.
torchvision==0.20.1
# Nothing here imports scipy, but transformers does whenever it is installed, and anything
# that pulls it in unpinned (scikit-learn does) gets a release built for numpy 2.x, which
# breaks transformers at import against the numpy<2 pin above. 1.16.3 is the newest that
# supports numpy 1.26.
scipy==1.16.3

# --- test / lint ---
pytest==8.3.4
